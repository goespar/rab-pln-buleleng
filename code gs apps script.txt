function doGet(e) {
  try {
    var action = e.parameter.action;
    var table = e.parameter.table;
    
    if (action === 'list' && table) {
      return createJsonResponse({ success: true, data: getTableData(table) });
    } else if (action === 'dashboard') {
      return createJsonResponse({ success: true, data: {} });
    }
    return createJsonResponse({ success: false, message: 'Action tidak valid.' });
  } catch (error) {
    return createJsonResponse({ success: false, message: error.toString() });
  }
}

function doPost(e) {
  try {
    var payload = JSON.parse(e.postData.contents);
    var action = payload.action || (e.parameter && e.parameter.action);
    var table = payload.table || (e.parameter && e.parameter.table);
    var rowData = payload.data || {};

    if (action === 'simpanRABTerkoreksi') {
      return createJsonResponse(
        simpanRABTerkoreksi(payload.pekerjaan_id, payload.dataKoreksi)
      );
    }

    if (action === 'cekLimitRealisasi') {
      return createJsonResponse(
        cekLimitRealisasi(payload.pekerjaan_id, payload.nilai_termin_baru, payload.realisasi_id)
      );
    }

    if (action === 'simpanMaterialBatch') {
      return createJsonResponse(
        simpanMaterialBatch(payload.dataMaterial)
      );
    }
    
    var ss = SpreadsheetApp.getActiveSpreadsheet();
    var sheet = ss.getSheetByName(table);
    if (!sheet) throw new Error("Sheet '" + table + "' tidak ditemukan.");

    // ACTION: CREATE (SIMPAN DATA BARU)
    if (action === 'create') {
      var headers = getHeaders(sheet);
      var newRow = [];
      var newId = "ID_" + new Date().getTime() + Math.floor(Math.random() * 1000);
      
      for (var i = 0; i < headers.length; i++) {
        var headerName = headers[i].toLowerCase().trim();
        if (headerName === 'id') newRow.push(newId);
        else if (headerName === 'created_at' || headerName === 'timestamp') newRow.push(new Date());
        else newRow.push(rowData[headerName] !== undefined ? rowData[headerName] : '');
      }
      sheet.appendRow(newRow);
      return createJsonResponse({ success: true, message: 'Disimpan.', id: newId });
    }
    
    // ACTION: UPDATE (EDIT DATA)
    else if (action === 'update') {
      var idToUpdate = rowData.id;
      if (!idToUpdate) throw new Error("ID tidak diberikan untuk update.");
      var data = sheet.getDataRange().getValues();
      var headers = data[0];
      var idIndex = headers.findIndex(h => h.toLowerCase().trim() === 'id');
      if (idIndex === -1) throw new Error("Kolom 'id' tidak ditemukan.");

      var updated = false;
      for (var r = 1; r < data.length; r++) {
        if (String(data[r][idIndex]) === String(idToUpdate)) {
          for (var c = 0; c < headers.length; c++) {
            var headerName = headers[c].toLowerCase().trim();
            if (headerName !== 'id' && headerName !== 'created_at' && rowData[headerName] !== undefined) {
              sheet.getRange(r + 1, c + 1).setValue(rowData[headerName]);
            }
          }
          updated = true;
          break;
        }
      }
      if (updated) return createJsonResponse({ success: true, message: 'Data diupdate.' });
      else throw new Error("Data ID tidak ditemukan.");
    }
    
    // ACTION: DELETE (HAPUS DATA)
    else if (action === 'delete') {
      var idToDelete = payload.id;
      var data = sheet.getDataRange().getValues();
      var headers = data[0];
      var idIndex = headers.findIndex(h => h.toLowerCase().trim() === 'id');
      
      var deleted = false;
      for (var r = data.length - 1; r > 0; r--) {
        if (String(data[r][idIndex]) === String(idToDelete)) {
          sheet.deleteRow(r + 1);
          deleted = true;
          break;
        }
      }
      if (deleted) return createJsonResponse({ success: true, message: 'Data dihapus.' });
      else throw new Error("Data ID tidak ditemukan.");
    }
    
    throw new Error("Action POST tidak dikenali.");
  } catch (error) {
    return createJsonResponse({ success: false, message: error.toString() });
  }
}

function doOptions(e) { return createJsonResponse({ success: true }); }

function getHeaders(sheet) {
  var lastCol = sheet.getLastColumn();
  return lastCol === 0 ? [] : sheet.getRange(1, 1, 1, lastCol).getValues()[0];
}

function getTableData(tableName) {
  var sheet = SpreadsheetApp.getActiveSpreadsheet().getSheetByName(tableName);
  if (!sheet || sheet.getLastRow() <= 1) return [];
  var data = sheet.getDataRange().getValues();
  var headers = data[0], result = [];
  for (var r = 1; r < data.length; r++) {
    var obj = {}, rowEmpty = true;
    for (var c = 0; c < headers.length; c++) {
      var headerName = headers[c].toLowerCase().trim();
      if (headerName) { obj[headerName] = data[r][c]; if (data[r][c] !== '') rowEmpty = false; }
    }
    if (!rowEmpty) result.push(obj);
  }
  return result;
}

function simpanMaterialBatch(dataMaterial) {
  var ss = SpreadsheetApp.getActiveSpreadsheet();
  var materialSheet = ss.getSheetByName('Material');
  if (!materialSheet) throw new Error("Sheet 'Material' tidak ditemukan.");
  if (!Array.isArray(dataMaterial) || dataMaterial.length === 0) {
    throw new Error('Basket material tidak boleh kosong.');
  }

  var headers = getHeaders(materialSheet);
  if (headers.length === 0) throw new Error("Header Sheet 'Material' belum tersedia.");
  var rows = dataMaterial.map(function (item) {
    var rowData = {};
    Object.keys(item || {}).forEach(function (key) {
      rowData[String(key).toLowerCase().trim()] = item[key];
    });
    rowData.id = 'ID_' + new Date().getTime() + Math.floor(Math.random() * 1000);
    if (rowData.created_at === undefined) rowData.created_at = new Date();
    return headers.map(function (header) {
      var headerName = String(header).toLowerCase().trim();
      return rowData[headerName] !== undefined ? rowData[headerName] : '';
    });
  });

  materialSheet.getRange(materialSheet.getLastRow() + 1, 1, rows.length, headers.length).setValues(rows);
  return {
    success: true,
    message: 'Material batch berhasil disimpan.',
    jumlah_disimpan: rows.length
  };
}

function simpanRABTerkoreksi(pekerjaan_id, dataKoreksi) {
  var ss = SpreadsheetApp.getActiveSpreadsheet();
  var rabSheet = ss.getSheetByName('RAB');
  var pekerjaanSheet = ss.getSheetByName('Pekerjaan');

  if (!rabSheet) throw new Error("Sheet 'RAB' tidak ditemukan.");
  if (!pekerjaanSheet) throw new Error("Sheet 'Pekerjaan' tidak ditemukan.");
  if (!pekerjaan_id) throw new Error('pekerjaan_id wajib diisi.');
  if (!Array.isArray(dataKoreksi) || dataKoreksi.length === 0) {
    throw new Error('Data koreksi RAB tidak boleh kosong.');
  }

  var rabData = rabSheet.getDataRange().getValues();
  var rabHeaders = getHeaders(rabSheet);
  var idIndex = cariIndexKolom(rabHeaders, ['id']);
  var pekerjaanIdIndex = cariIndexKolom(rabHeaders, ['pekerjaan_id']);
  var versiIndex = cariIndexKolom(rabHeaders, ['versi_rab']);
  if (idIndex === -1 || pekerjaanIdIndex === -1 || versiIndex === -1) {
    throw new Error("Sheet RAB wajib memiliki kolom 'id', 'pekerjaan_id', dan 'versi_rab'.");
  }

  var correctedRows = [];
  for (var r = 1; r < rabData.length; r++) {
    if (String(rabData[r][pekerjaanIdIndex]) === String(pekerjaan_id) &&
        String(rabData[r][versiIndex] || '').trim().toLowerCase() === 'terkoreksi') {
      correctedRows.push(r + 1);
    }
  }

  var rabRows = dataKoreksi.map(function (item) {
    var rowData = {};
    Object.keys(item || {}).forEach(function (key) {
      rowData[String(key).toLowerCase().trim()] = item[key];
    });

    rowData.id = rowData.id || ('ID_' + new Date().getTime() + Math.floor(Math.random() * 1000));
    rowData.pekerjaan_id = pekerjaan_id;
    rowData.versi_rab = 'Terkoreksi';
    if (rowData.created_at === undefined) rowData.created_at = new Date();

    return rabHeaders.map(function (header) {
      var headerName = String(header).toLowerCase().trim();
      return rowData[headerName] !== undefined ? rowData[headerName] : '';
    });
  });

  var commonRows = Math.min(correctedRows.length, rabRows.length);
  for (var i = 0; i < commonRows; i++) {
    rabSheet.getRange(correctedRows[i], 1, 1, rabHeaders.length).setValues([rabRows[i]]);
  }
  for (var deleteIndex = correctedRows.length - 1; deleteIndex >= rabRows.length; deleteIndex--) {
    rabSheet.deleteRow(correctedRows[deleteIndex]);
  }
  if (rabRows.length > commonRows) {
    rabSheet.getRange(rabSheet.getLastRow() + 1, 1, rabRows.length - commonRows, rabHeaders.length)
      .setValues(rabRows.slice(commonRows));
  }

  var pekerjaanData = pekerjaanSheet.getDataRange().getValues();
  var pekerjaanHeaders = pekerjaanData[0];
  var idIndex = cariIndexKolom(pekerjaanHeaders, ['pekerjaan_id', 'id']);
  var statusIndex = cariIndexKolom(pekerjaanHeaders, ['status']);

  if (idIndex === -1) throw new Error("Kolom 'pekerjaan_id' atau 'id' tidak ditemukan.");
  if (statusIndex === -1) throw new Error("Kolom 'status' tidak ditemukan.");

  for (var i = 1; i < pekerjaanData.length; i++) {
    if (String(pekerjaanData[i][idIndex]) === String(pekerjaan_id)) {
      pekerjaanSheet.getRange(i + 1, statusIndex + 1).setValue('Menunggu PA');
      return {
        success: true,
        message: 'RAB terkoreksi berhasil disimpan.',
        pekerjaan_id: pekerjaan_id,
        versi_rab: 'Terkoreksi',
        status_pekerjaan: 'Menunggu PA',
        jumlah_detail: rabRows.length
      };
    }
  }

  throw new Error('Pekerjaan dengan ID tersebut tidak ditemukan.');
}

function cekLimitRealisasi(pekerjaan_id, nilai_termin_baru, realisasi_id) {
  var ss = SpreadsheetApp.getActiveSpreadsheet();
  var kontrakSheet = ss.getSheetByName('Kontrak');
  var realisasiSheet = ss.getSheetByName('Realisasi');

  if (!kontrakSheet) throw new Error("Sheet 'Kontrak' tidak ditemukan.");
  if (!realisasiSheet) throw new Error("Sheet 'Realisasi' tidak ditemukan.");
  if (!pekerjaan_id) throw new Error('pekerjaan_id wajib diisi.');

  var nilaiTerminBaru = parseNilaiAngka(nilai_termin_baru);
  if (nilaiTerminBaru < 0) throw new Error('Nilai termin baru tidak boleh negatif.');

  var kontrakData = kontrakSheet.getDataRange().getValues();
  var kontrakHeaders = kontrakData[0];
  var kontrakIdIndex = cariIndexKolom(kontrakHeaders, ['pekerjaan_id', 'id_pekerjaan']);
  var nilaiKontrakIndex = cariIndexKolom(kontrakHeaders, ['nilai_kontrak', 'total_kontrak']);

  if (kontrakIdIndex === -1 || nilaiKontrakIndex === -1) {
    throw new Error("Kolom 'pekerjaan_id' atau 'nilai_kontrak' tidak ditemukan di sheet Kontrak.");
  }

  var nilaiKontrak = 0;
  for (var i = 1; i < kontrakData.length; i++) {
    if (String(kontrakData[i][kontrakIdIndex]) === String(pekerjaan_id)) {
      nilaiKontrak += parseNilaiAngka(kontrakData[i][nilaiKontrakIndex]);
    }
  }
  if (nilaiKontrak <= 0) throw new Error('Kontrak tidak ditemukan atau nilai kontrak tidak valid.');

  var realisasiData = realisasiSheet.getDataRange().getValues();
  var realisasiHeaders = realisasiData[0];
  var realisasiIdIndex = cariIndexKolom(realisasiHeaders, ['pekerjaan_id', 'id_pekerjaan']);
  var realisasiRowIdIndex = cariIndexKolom(realisasiHeaders, ['id']);
  var nilaiRealisasiIndex = cariIndexKolom(realisasiHeaders, ['nilai_termin', 'nilai_realisasi', 'nilai']);

  if (realisasiIdIndex === -1 || nilaiRealisasiIndex === -1) {
    throw new Error("Kolom 'pekerjaan_id' atau nilai realisasi tidak ditemukan di sheet Realisasi.");
  }

  var totalRealisasiSebelumnya = 0;
  for (var j = 1; j < realisasiData.length; j++) {
    if (String(realisasiData[j][realisasiIdIndex]) === String(pekerjaan_id) &&
      (!realisasi_id || realisasiRowIdIndex === -1 || String(realisasiData[j][realisasiRowIdIndex]) !== String(realisasi_id))) {
      totalRealisasiSebelumnya += parseNilaiAngka(realisasiData[j][nilaiRealisasiIndex]);
    }
  }

  var totalSetelahTermin = totalRealisasiSebelumnya + nilaiTerminBaru;
  if (totalSetelahTermin > nilaiKontrak) {
    return {
      success: false,
      error: true,
      message: 'Nilai realisasi melebihi nilai kontrak.',
      nilai_kontrak: nilaiKontrak,
      total_realisasi_sebelumnya: totalRealisasiSebelumnya,
      nilai_termin_baru: nilaiTerminBaru,
      total_setelah_termin: totalSetelahTermin,
      sisa_kontrak: nilaiKontrak - totalRealisasiSebelumnya
    };
  }

  return {
    success: true,
    error: false,
    message: 'Nilai realisasi masih dalam batas kontrak.',
    nilai_kontrak: nilaiKontrak,
    total_realisasi_sebelumnya: totalRealisasiSebelumnya,
    nilai_termin_baru: nilaiTerminBaru,
    total_setelah_termin: totalSetelahTermin,
    sisa_kontrak: nilaiKontrak - totalSetelahTermin
  };
}

function cariIndexKolom(headers, namaKolom) {
  return headers.findIndex(function (header) {
    return namaKolom.indexOf(String(header).toLowerCase().trim()) !== -1;
  });
}

function parseNilaiAngka(nilai) {
  if (typeof nilai === 'number') return nilai;
  if (nilai === null || nilai === undefined || nilai === '') return 0;

  var teks = String(nilai).replace(/\s/g, '').replace(/Rp/gi, '');
  if (teks.indexOf(',') !== -1) {
    teks = teks.replace(/\./g, '').replace(',', '.');
  } else {
    teks = teks.replace(/\./g, '');
  }

  var angka = Number(teks);
  return isNaN(angka) ? 0 : angka;
}

function createJsonResponse(responseObject) {
  return ContentService.createTextOutput(JSON.stringify(responseObject)).setMimeType(ContentService.MimeType.JSON);
}
